    CONSTANTE ISOSCELES <- 5

    // DECLARACIÓN DE VARIABLES PRINCIPALES
    ENTERO opcion <- 0
    ENTERO figuraActual <- NINGUNA
    ENTERO A <- 0
    ENTERO B <- 0
    ENTERO C <- 0
    REAL areaCalculada <- 0.0

    // MENÚ INTERACTIVO
    HACER

        ESCRIBIR "Seleccione una opción: "

                LEER B

                figuraActual <- DeterminarFigura(A, B, C)

            CASO 2:
                SI figuraActual == NINGUNA ENTONCES
                    ESCRIBIR "Primero deberá determinar la figura en la opción 1"
                SINO

                        CASO CUADRADO:
                            ESCRIBIR "El área del CUADRADO es ", areaCalculada

            CASO 3:

        FIN SEGUN

    MIENTRAS opcion != 3

FIN

// FUNCIÓN: DeterminarFigura
FUNCIÓN DeterminarFigura(ENTERO a, ENTERO b, ENTERO c) RETORNA ENTERO
    SI a == 0 Y b == 0 Y c == 0 ENTONCES
        RETORNAR NINGUNA
    SINO SI b == 0 Y c == 0 ENTONCES
        SI a > 0 ENTONCES
            RETORNAR CIRCULO
        SINO
            RETORNAR NINGUNA
        FIN SI
    SINO SI c == 0 ENTONCES
        SI a <= 0 O b <= 0 ENTONCES
            RETORNAR NINGUNA
        FIN SI
        
        SI a == b ENTONCES
            RETORNAR CUADRADO
        SINO
            RETORNAR RECTANGULO
        FIN SI
    SINO
        // Validación de lados positivos y desigualdad triangular
        SI a <= 0 O b <= 0 O c <= 0 O (a + b <= c) O (a + c <= b) O (b + c <= a) ENTONCES
            RETORNAR NINGUNA
        SINO SI a == b Y b == c ENTONCES
            RETORNAR EQUILATERO
        SINO SI a == b O a == c O b == c ENTONCES
            RETORNAR ISOSCELES
        SINO
            RETORNAR ESCALENO
        FIN SI
    FIN SI
FIN FUNCIÓN


// FUNCIÓN: CalcularArea
FUNCIÓN CalcularArea(ENTERO figura, ENTERO a, ENTERO b, ENTERO c) RETORNA REAL
    REAL semiperimetro <- 0.0
    REAL resultadoArea <- 0.0

    SEGUN figura HACER
        CASO CUADRADO:
            RETORNAR a * a

        CASO RECTANGULO:
            RETORNAR a * b

        CASO CIRCULO:
            RETORNAR PI * (a * a)

        CASO EQUILATERO:
        CASO ISOSCELES:
        CASO ESCALENO:
            // Fórmula de Herón
            semiperimetro <- (a + b + c) / 2.0
            resultadoArea <- RAIZ_CUADRADA(semiperimetro * (semiperimetro - a) * (semiperimetro - b) * (semiperimetro - c))
            RETORNAR resultadoArea

        DE OTRO MODO:
            RETORNAR 0.0
    FIN SEGUN
FIN FUNCIÓN
